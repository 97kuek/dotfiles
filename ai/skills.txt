# ClaudeとCodexで共通に使う、GitHubで配布されているスキル。
# ai/install.sh がリポジトリを取得し、~/.agents/skills/<名前> にリンクする。
# Claude Codeへは、そこから ~/.claude/skills/<名前> にリンクする。
#
# 形式: <GitHubのowner/repo>  <リポジトリ内のスキルのディレクトリ>
#   スキルの名前はディレクトリ名になる。
#
# 1つのスキルだけを入れたいときはここに書く。
# 複数のスキルやフックをまとめて配っているものは plugins.txt に書く。
# 一覧と内容は docs/skills.md にまとめている。追加や削除をしたら合わせて更新する。
#
# #から行末まではコメント。空行は無視される。

# Anthropic
anthropics/skills         skills/frontend-design        # 見た目に個性のあるUIを作る
anthropics/skills         skills/doc-coauthoring        # 仕様書や提案書を対話しながら書く
anthropics/skills         skills/docx                   # Word
anthropics/skills         skills/pptx                   # PowerPoint
anthropics/skills         skills/xlsx                   # Excel
anthropics/skills         skills/pdf                    # PDF

# Vercel
vercel-labs/agent-skills  skills/web-design-guidelines  # UIをWeb Interface Guidelinesでレビューする
vercel-labs/skills        skills/find-skills            # 公開されているスキルを探す

# HTMLの図（ロゴ入りの構成図など）
# effective-htmlはスキル同士が参照し合う。html が振り分け、html-diagram が図を作り、
# design-artifact が見た目の方向性を決める。html-diagram だけでは自動で使われない。
plannotator/effective-html  skills/html                 # HTMLで作るものの入口。図は html-diagram に回す
plannotator/effective-html  skills/html-diagram         # 構成図・シーケンス図・状態遷移などをHTMLで作る
plannotator/effective-html  skills/design-artifact      # 配色・フォント・レイアウトの方向性を決める
glincker/thesvg             skills/thesvg               # サービスやブランドのロゴ（SVG）を取ってくる
